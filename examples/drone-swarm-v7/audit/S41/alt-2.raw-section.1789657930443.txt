// ===== The replicable element (CV-16) =====

#Member part def SurveillanceDrone {
    doc /* The replicable element: one of the 12 identical surveillance drones. Alternative 2: most Coordination functions are decided on board, over the mesh, rather than by the ground. Carries its own #Mode and #State machines (CV-16, CV-18). */

    out port meshOut : Common::MeshPort;
    in port meshIn : ~Common::MeshPort;

    part meshRadio : MeshRadio {
        doc /* This drone's mesh radio hardware, carrying meshOut/meshIn traffic. */
    }
    part groundRadio : GroundLinkRadio {
        doc /* This drone's ground-link radio hardware. Present on every drone so any drone can relay (RelayLink); only the currently relaying drone's link is live. */
    }
    part storedMaps : StoredMapsStore {
        doc /* This drone's local copy of reference imagery for map-matching navigation. */
    }

    variation part navigationFallback : StoredMapMatchingNav {
        doc /* Variation point: which technique this drone uses to navigate without GNSS. */
        variant part mapMatching : StoredMapMatchingNav;
        variant part deadReckoning : DeadReckoningNav;
    }

    attribute flightEnduranceMinutesBudget = 40 [min];
    require constraint {
        doc /* memberFlightEnduranceMinutes budget: this drone flies for about 40 minutes per sortie. */
        flightEnduranceMinutesBudget >= 40 [min]
    }

    attribute onboardClassificationCostBudget = 300;
    require constraint {
        doc /* ComputingCostCap: on-board classification computing costs no more than 300 USD per drone. USD is not an ISQ unit, so the budget is carried unitless; the currency is stated here. */
        onboardClassificationCostBudget <= 300
    }

    // ---- Coordination decided on board in this alternative (7 of 8 functions; CV-16 architecture decision) ----
    part handOverSectorFn : SurveillanceDroneSwarm::LA::HandOverSector {
        doc /* On-board usage of #Coordination HandOverSector: this drone negotiates sector handover directly with its mesh peers. */
    }
    part rotateRechargeFn : SurveillanceDroneSwarm::LA::RotateRecharge {
        doc /* On-board usage of #Coordination RotateRecharge: this drone agrees its recharge slot with the fleet over the mesh. */
    }
    part redistributeCoverageFn : SurveillanceDroneSwarm::LA::RedistributeCoverage {
        doc /* On-board usage of #Coordination RedistributeCoverage: this drone re-spreads to cover a lost peer's sector without waiting for the ground. */
    }
    part deconflictMembersFn : SurveillanceDroneSwarm::LA::DeconflictMembers {
        doc /* On-board usage of #Coordination DeconflictMembers: this drone deconflicts flight paths with nearby peers directly. */
    }
    part relayLinkFn : SurveillanceDroneSwarm::LA::RelayLink {
        doc /* On-board usage of #Coordination RelayLink: this drone can relay another drone's traffic toward the ground link. */
    }
    part correlateTracksFn : SurveillanceDroneSwarm::LA::CorrelateTracks {
        doc /* On-board usage of #Coordination CorrelateTracks: this drone correlates its tracks with peers' tracks over the mesh. */
    }
    part handOverTrackFn : SurveillanceDroneSwarm::LA::HandOverTrack {
        doc /* On-board usage of #Coordination HandOverTrack: this drone hands a track to the peer best placed to continue it. */
    }

    // ---- Other per-member behaviour (typing pending: LA text was not visible to this layer, see todos) ----
    part detectAndClassifyFn {
        doc /* Realises DetectClassifyAndReport on board: detects moving objects, classifies them or reports Unknown, with position and time. */
    }
    part navigateFn {
        doc /* Realises NavigateWithoutSatellitePositioning: matches camera frames against storedMaps when GNSS is unavailable (mitigates PositionDriftHazard). */
    }
    part stayWithinClearanceFn {
        doc /* Realises StayWithinClearedAirspace: enforces AirspaceClearance and forces a landing on excess wind or geofence breach. */
    }
    part triageFn {
        doc /* Realises TriageReports on board: ranks and de-duplicates this drone's reports before they reach the mesh/ground, reducing operator load. */
    }
    part decisionRecordFn {
        doc /* Realises KeepDecisionRecord: records, per report, its frames, classifier version, the reporting threshold in force and (once returned) the acknowledger. */
    }
    part linkGapBufferFn {
        doc /* Realises OperateThroughLinkGap: holds this drone's reports when the ground link is down and delivers them once it returns, with none lost. */
    }

    // ---- State machines ----
    #State state def MemberState {
        doc /* Flight/operational state of this drone. Carries RecallWins, GeofenceBreachEndsWatch and ReturnsWhenIsolated (CV-18). Isolated (cut off from the mesh) is a flight state here so recovery-to-Landed can be checked across it. */

        initial start;
        state Landed;
        state Launching;
        state Watching;
        state Isolated;
        state Recalled;
        state OutsideClearance;
        state Landing;

        transition start -> Landed;
        transition Landed -> Launching;
        transition Launching -> Watching;
        transition Watching -> Isolated;
        transition Isolated -> Watching;
        transition Isolated -> Landing;
        transition Watching -> Recalled;
        transition Watching -> OutsideClearance;
        transition Recalled -> Landing;
        transition OutsideClearance -> Landing;
        transition Landing -> Landed;

        @SysproseVerification::PropertyPattern {
            doc /* RecallWins: once a drone is recalled, it does not watch again until it has landed. */
            attribute pattern = "absence";
            attribute scope = "between";
            attribute q = "state Recalled";
            attribute r = "state Landed";
            attribute p = "state Watching";
        }
        @SysproseVerification::PropertyPattern {
            doc /* GeofenceBreachEndsWatch: once a drone is outside its cleared airspace, it does not watch until it has landed. */
            attribute pattern = "absence";
            attribute scope = "between";
            attribute q = "state OutsideClearance";
            attribute r = "state Landed";
            attribute p = "state Watching";
        }
        @SysproseVerification::PropertyPattern {
            doc /* ReturnsWhenIsolated: from any situation a drone can be in, it can always get back to landed; an isolated drone included. */
            attribute pattern = "recovery";
            attribute scope = "globally";
            attribute p = "state Landed";
        }
    }

    #Mode state def MemberMode {
        doc /* On-board condition mode of this drone: GNSS/sensing degradation and mesh admission status. Carries QuarantinedStaysOut (CV-18). */

        initial modeStart;
        state Nominal;
        state NavigationDegraded;
        state SensingDegraded;
        state Quarantined;
        state Readmitting;
        state Readmitted;

        transition modeStart -> Nominal;
        transition Nominal -> NavigationDegraded;
        transition NavigationDegraded -> Nominal;
        transition Nominal -> SensingDegraded;
        transition SensingDegraded -> Nominal;
        transition Nominal -> Quarantined;
        transition Quarantined -> Readmitting;
        transition Readmitting -> Readmitted;
        transition Readmitted -> Nominal;

        @SysproseVerification::PropertyPattern {
            doc /* QuarantinedStaysOut: once a drone is quarantined, it takes no part in coordination until it has been readmitted. */
            attribute pattern = "absence";
            attribute scope = "between";
            attribute q = "state Quarantined";
            attribute r = "state Readmitted";
            attribute p = "state Nominal";
        }
    }

    perform action memberStateMachine : MemberState {
        doc /* Executes this drone's flight/operational state machine. */
    }
    perform action memberModeMachine : MemberMode {
        doc /* Executes this drone's condition-mode state machine. */
    }
}

// ===== Fleet-level configuration (CV-16) =====

#Configuration state def FleetConfiguration {
    doc /* Fleet-wide coordination configuration. MeshDegraded is the brief-named mode of the fleet when a share of inter-drone links is jammed. */
    initial fleetStart;
    state Nominal;
    state MeshDegraded;
    transition fleetStart -> Nominal;
    transition Nominal -> MeshDegraded;
    transition MeshDegraded -> Nominal;
}

#Node part def GroundStation {
    doc /* The operations centre's physical ground node: console, C2 software and the mesh-admission authority. In alternative 2 this is a thin node: it keeps every #C2 function and only the AdmitMember #Coordination function; the rest of Coordination runs on board the fleet. */

    port groundLinkPortA : ~Common::GroundLinkPort;
    port groundLinkPortB : ~Common::GroundLinkPort;

    // ---- Command and control: every #C2 function keeps a ground usage in every alternative ----
    part taskMissionFn : SurveillanceDroneSwarm::LA::TaskMission {
        doc /* Ground usage of #C2 TaskMission: the duty controller sets the area of interest, sectors and mission window. */
    }
    part commandSupervisionFn : SurveillanceDroneSwarm::LA::CommandSupervision {
        doc /* Ground usage of #C2 CommandSupervision: sets priorities, the close-look sector and the reporting threshold. */
    }
    part recallAndLandFn : SurveillanceDroneSwarm::LA::RecallAndLand {
        doc /* Ground usage of #C2 RecallAndLand: issues swarm-wide or single-drone recall-and-land. */
    }
    part presentStatusPictureFn : SurveillanceDroneSwarm::LA::PresentStatusPicture {
        doc /* Ground usage of #C2 PresentStatusPicture: composes the status picture from member reports for the duty controller. */
    }
    part acknowledgeReportFn : SurveillanceDroneSwarm::LA::AcknowledgeReport {
        doc /* Ground usage of #C2 AcknowledgeReport: the duty controller's acknowledgement of each detection report. */
    }

    // ---- The one #Coordination function this alternative keeps on the ground ----
    part admitMemberFn : SurveillanceDroneSwarm::LA::AdmitMember {
        doc /* Ground-hosted #Coordination AdmitMember: the ground station is the admission authority, deciding whether a mesh join request is a genuine fleet member (mitigates SpoofedMemberHazard). */
    }

    part classifierVersionFn {
        doc /* Realises ManageClassifierVersion from the ground: pushes and tracks the single classifier version every drone must run (mitigates MixedClassifierVersionsHazard). */
    }

    #State state def GroundOperatingState {
        doc /* Ground station operating state. */
        initial gsStart;
        state Idle;
        state Supervising;
        transition gsStart -> Idle;
        transition Idle -> Supervising;
        transition Supervising -> Idle;
    }
    #Mode state def GroundLinkMode {
        doc /* Ground-side link mode: whether the ground link radio currently has a connected drone. */
        initial glStart;
        state LinkUp;
        state LinkDown;
        transition glStart -> LinkDown;
        transition LinkDown -> LinkUp;
        transition LinkUp -> LinkDown;
    }
    perform action groundOperatingStateMachine : GroundOperatingState {
        doc /* Executes the ground station's operating-state machine. */
    }
    perform action groundLinkModeMachine : GroundLinkMode {
        doc /* Executes the ground station's link-mode machine. */
    }

    attribute operatorAlertsPerHourBudget = 20;
    attribute operatorSecondsPerAlertBudget = 30 [s];
    require constraint {
        doc /* OperatorWorkloadLimit: at most 20 alerts per hour reach the duty controller (mitigates OperatorOverloadHazard). */
        operatorAlertsPerHourBudget <= 20
    }
    require constraint {
        doc /* OperatorWorkloadLimit: at least 30 seconds is available to judge each alert. */
        operatorSecondsPerAlertBudget >= 30 [s]
    }
}

package Hazards {
    #Hazard requirement GroundLinkRadioFailureHazard {
        doc /* PA-added: the physical ground link radio on the relaying member, or at the ground station, fails outright rather than merely degrading, losing the ground link regardless of RadioInterference conditions. */
    }
    #Hazard requirement RechargeContentionHazard {
        doc /* PA-added: more drones need the recharge point at once than the ground crew can turn around within a rotation cycle, so a scheduled handover has no relieving drone. */
    }
}
