package PA {
  doc /* Physical Architecture, alternative 2 of 2 (drone-centric). The fleet's twelve identical airframes host the LA-defined SurveillanceDrone member behaviour by nesting. Ground-facing tasking, supervisory command, recall, status and acknowledgement functions keep a ground usage, as required of every alternative, but the member-to-member coordination functions run mostly onboard: what every member does for itself is allocated to the fleet, and what one member decides for the others is allocated to an elected onboard representative rather than to the ground — six of the eight #Coordination functions, well over half, sit on a member here, the opposite choice from a ground-centred alternative 1. */

  // ===== Physical connection media =====
  connection def RFMeshLink {
    doc /* The physical RF medium that types the mesh link between two representative drones; carried in hardware by each drone's onboard MeshRadioUnit #Node part, never modelled as an actor. */
  }

  connection def RFGroundLink {
    doc /* The physical RF medium that types the ground-station link between the onboard coordinator drone and the ground station; not always available, per GroundStationLink. */
  }

  // ===== Physical node part definitions =====
  #Node part def MeshRadioUnit {
    doc /* The physical mesh radio hardware nested inside every drone node; the tangible bearer of the RFMeshLink connection between members. A hardware fault here removes a member's coordination bearer even without interference; see Hazards::MeshRadioHardwareFailureHazard. */
  }

  #Node part def GroundRadioUnit {
    doc /* The physical ground-link radio hardware nested in the coordinator drone node and in the ground station node; the tangible bearer of the RFGroundLink connection. */
    out port linkOut : Common::GroundLinkPort;
    in port linkIn : ~Common::GroundLinkPort;
  }

  #Node part def DroneNode {
    doc /* One physical drone airframe and its onboard compute: hosts the LA member behaviour by nesting (CV-06) and carries its own mesh radio hardware. */
    part member : SurveillanceDroneSwarm::LA::SurveillanceDrone;
    part meshRadio : MeshRadioUnit;
  }

  #Node #Variant part def CoordinatorDroneNode :> DroneNode {
    doc /* Specializes DroneNode with an additional ground-link radio. Alternative 2's own decision: ground-facing traffic and the deciding share of coordination concentrate on one elected member rather than every drone polling the ground station, or a ground coordinator deciding for all as in alternative 1. Tagged #Variant because which physical airframe plays this role can change on admitMember/handOverSector re-election. */
    part groundRadio : GroundRadioUnit;
  }

  #Node part def GroundStationNode {
    doc /* The fixed ground station: hosts the LA ground-station logical component by nesting (CV-06) and carries the ground-link radio hardware. Retains a usage of every #C2 function in both alternatives. */
    part groundControl : SurveillanceDroneSwarm::LA::GroundStation;
    part groundRadio : GroundRadioUnit;
  }

  #Node part def RecoveryPointNode {
    doc /* The physical recovery point a drone returns to autonomously once isolated from the swarm, realizing the brief's RecoveryPoint resource and the ReturnsWhenIsolated rule already carried on the member's own state machine. */
  }

  #Node part def RechargeBay {
    doc /* The ground point and crew where a battery pack is swapped rather than charged in place, giving the fixed 20-minute turnaround; the swap contacts wear with repeated cycles, see Hazards::BatteryContactWearHazard. */
    part battery : Common::BatteryPack;
  }

  // ===== The population, surviving into the physical architecture (CV-16) =====
  part fleet : DroneNode [12] {
    doc /* The physical population: twelve identical drone nodes, each nesting one usage of the LA #Member part def SurveillanceDrone. Never collapsed to one instance. */
  }

  part droneA : CoordinatorDroneNode {
    doc /* Representative fleet member A: the elected onboard coordinator for this architecture's member-held coordination functions and the swarm's ground-link gateway. */
  }

  part droneB : DroneNode {
    doc /* Representative fleet member B: an ordinary fleet member, exhibited here to carry the peer mesh link to droneA. */
  }

  connection meshPeerLink : RFMeshLink connect droneA.member.meshOut to droneB.member.meshIn {
    doc /* The link between two members of the population: a connection between two representatives, typed at PA by the connection def of the medium (RFMeshLink) that carries member-to-member coordination traffic, independent of the ground link. */
  }

  part groundStation : GroundStationNode {
    doc /* The single physical ground station. */
  }

  part recoveryPoint : RecoveryPointNode {
    doc /* The physical recovery point, reachable independently of the mesh or ground link. */
  }

  part rechargeBay : RechargeBay {
    doc /* The physical recharge bay serving the fleet's rotation. */
  }

  connection groundLink : RFGroundLink connect droneA.groundRadio.linkOut to groundStation.groundRadio.linkIn {
    doc /* The physical ground-station link, carried by the coordinator drone's ground radio rather than by every member; the architecture's own choice of where the ground-facing gateway sits. */
  }

  // ===== Function allocation: the architecture decision (CV-16) =====
  allocate SurveillanceDroneSwarm::OA::TaskSurveillanceMission to groundStation.groundControl;
  allocate SurveillanceDroneSwarm::OA::IssueSupervisoryCommand to groundStation.groundControl;
  allocate SurveillanceDroneSwarm::OA::RecallToLand to groundStation.groundControl;
  allocate SurveillanceDroneSwarm::OA::PresentStatusPicture to groundStation.groundControl;
  allocate SurveillanceDroneSwarm::OA::AcknowledgeDetectionReport to groundStation.groundControl;

  allocate SurveillanceDroneSwarm::OA::DeconflictFlight to fleet;
  allocate SurveillanceDroneSwarm::OA::RelayThroughNeighbour to fleet;

  allocate SurveillanceDroneSwarm::OA::HandOverSector to droneA.member;
  allocate SurveillanceDroneSwarm::OA::RotateRecharge to droneA.member;
  allocate SurveillanceDroneSwarm::OA::RedistributeCoverage to droneA.member;
  allocate SurveillanceDroneSwarm::OA::CorrelateTracks to droneA.member;
  allocate SurveillanceDroneSwarm::OA::HandOverTrack to droneA.member;
  allocate SurveillanceDroneSwarm::OA::AdmitMember to droneA.member;

  // ===== Resource budgets restated as local valued attributes and require constraints (CV-17) =====
  part def PhysicalBudgets {
    doc /* Restates the brief's fixed fleet-and-endurance numbers as valued attributes of this layer, because a constraint body cannot reach Common; used below to derive the population-share estimates and to state resource budgets as require constraints. */

    attribute flightDuration : ISQ::DurationValue = 2400 [s] {
      doc /* The fixed 40-minute (2400 s) flight endurance per sortie, from FleetAndEnduranceBudget. */
    }
    attribute turnaroundDuration : ISQ::DurationValue = 1200 [s] {
      doc /* The fixed 20-minute (1200 s) battery-swap turnaround, from FleetAndEnduranceBudget. */
    }
    attribute fleetCount : Real = 12 {
      doc /* The fixed twelve drones, from FleetAndEnduranceBudget. */
    }
    attribute sensorFootprintAreaSqm : Real = 3000000 {
      doc /* About 3.0 km^2 (3,000,000 m^2) watched from altitude by one drone, from FleetAndEnduranceBudget; the unit is dropped to a plain number here and kept in this doc, since km^2 is not a library token. */
    }
    attribute areaOfInterestSqm : Real = 25000000 {
      doc /* Up to 25 km^2 (25,000,000 m^2) area of interest, fixed by the brief; unit dropped here for the same reason. */
    }

    require constraint FlightEnduranceBudget {
      doc /* The airframe/battery design must deliver at least the fixed flight endurance the coverage arithmetic depends on. */
      flightDuration >= 2400 [s]
    }
    require constraint GroundTurnaroundBudget {
      doc /* The recharge bay's battery-swap turnaround must not exceed the fixed 20-minute budget the airborne-share arithmetic depends on. */
      turnaroundDuration <= 1200 [s]
    }
    require constraint SensorFootprintBudget {
      doc /* Each drone's sensor at watch altitude must cover at least the fixed footprint the coverage arithmetic depends on. */
      sensorFootprintAreaSqm >= 3000000
    }
  }

  part physicalBudgets : PhysicalBudgets {
    doc /* This architecture's instance of the physical budgets. */
  }

  // ===== Worst-case estimates of the measures of effectiveness (CV-17) =====
  part def AlternativeTwoEstimates {
    doc /* Alternative 2's worst-case estimate for each measure of effectiveness, with its basis; the population-share measures are derived from physicalBudgets rather than stated as a literal, per CV-17's checked case. */

    #Estimate attribute areaUnderWatchFraction :> Common::areaUnderWatchFraction {
      doc /* Left without a literal: derived below, because a share held "at any moment" cannot exceed what the fleet's airborne share and per-member footprint actually tile over the area of interest. */
    }
    #Estimate attribute coverageUnderMeshJammingFraction :> Common::coverageUnderMeshJammingFraction {
      doc /* Left without a literal: derived below from areaUnderWatchFraction and the onboard-coordination resilience factor. */
    }
    #Estimate attribute watchEnduranceHoursWithoutPeople :> Common::watchEnduranceHoursWithoutPeople = 12 {
      doc /* Basis: the fixed 40/20-minute duty cycle rotates through all twelve drones indefinitely without operator intervention as long as the recharge bay and classifier release stay constant; 12 h is the architect's working target per the brief's own note, to be confirmed with the customer. */
    }
    #Estimate attribute reportDeliveryLatencySeconds :> Common::reportDeliveryLatencySeconds = 45 {
      doc /* Basis: onboard correlation at the coordinator drone plus one mesh relay hop before the ground link adds latency against a direct per-drone ground link; worst case 45 s, inside the 60 s target. */
    }
    #Estimate attribute positionErrorAfterGnssLossMetres :> Common::positionErrorAfterGnssLossMetres = 45 {
      doc /* Basis: camera navigation against StoredMaps over the fixed 10-minute GNSS outage budget; a navigation figure this architecture's coordination choice does not change. */
    }
    #Estimate attribute reportHoldDurationMinutes :> Common::reportHoldDurationMinutes = 35 {
      doc /* Basis: the coordinator drone's onboard store holds reports through a ground-link gap and reconciles once the link returns; worst case 35 min, above the 30 min target. */
    }
    #Estimate attribute missedDetectionFraction :> Common::missedDetectionFraction = 0.08 {
      doc /* Basis: classifier performance at the reporting threshold in force; an architecture-neutral figure carried from LA. */
    }
    #Estimate attribute falseAlarmsPerHour :> Common::falseAlarmsPerHour = 1.5 {
      doc /* Basis: classifier performance at the reporting threshold in force; an architecture-neutral figure carried from LA. */
    }
    #Estimate attribute alertsReachingOperatorPerHour :> Common::alertsReachingOperatorPerHour = 18 {
      doc /* Basis: triage and duplicate-merge ahead of the ground console keep the operator within the fixed 20-per-hour capacity. */
    }
    #Estimate attribute acknowledgedReportsThatMatterFraction :> Common::acknowledgedReportsThatMatterFraction = 0.82 {
      doc /* Basis: triage ranking at the ground console surfaces what matters first; worst case 0.82, above the 0.8 target. */
    }
    #Estimate attribute coverageDropAfterMemberLossFraction :> Common::coverageDropAfterMemberLossFraction = 0.20 {
      doc /* Basis: redistributeCoverage is decided onboard by the coordinator drone without waiting on the ground link, so re-spreading after a loss is faster than a ground-decided scheme; worst case 0.20, inside the 0.25 target. */
    }
    #Estimate attribute onboardClassifierCostUsdPerDrone :> Common::onboardClassifierCostUsdPerDrone = 280 {
      doc /* Basis: the onboard compute board priced for the chosen classifier release; worst case 280 USD per drone, inside the 300 USD target. */
    }

    attribute meshJammingResilienceFactor : Real = 0.85 {
      doc /* The share of areaUnderWatchFraction this architecture retains with half the mesh links jammed, because coordination is carried by the onboard representative and by every member's own deconfliction/relay behaviour rather than solely over the ground link; a valued attribute of this layer, not a claim on Common. */
    }

    assert constraint {
      doc /* CV-17 derivation: areaUnderWatchFraction cannot exceed what the fleet's airborne share and each member's footprint actually tile over the area of interest. */
      areaUnderWatchFraction == (physicalBudgets.fleetCount * (physicalBudgets.flightDuration / (physicalBudgets.flightDuration + physicalBudgets.turnaroundDuration)) * physicalBudgets.sensorFootprintAreaSqm) / physicalBudgets.areaOfInterestSqm
    }
    assert constraint {
      doc /* Mesh-jamming coverage derives from the same watch share, discounted by the onboard-coordination resilience factor above. */
      coverageUnderMeshJammingFraction == areaUnderWatchFraction * meshJammingResilienceFactor
    }
  }

  part alternativeTwoEstimates : AlternativeTwoEstimates {
    doc /* This architecture's instance of the estimates above. */
  }

  // ===== Hazards this layer adds (safety option) =====
  package Hazards {
    doc /* Hazards introduced by alternative 2's own physical technology choices: concentrating coordination on one onboard node, the mesh radio hardware itself, and the physical battery-swap mechanism. None restates a hazard named above. */

    #Hazard requirement LeaderNodeLossHazard {
      doc /* Concentrating six of the eight coordination functions on one elected representative drone means losing that specific airframe stalls handOverSector, rotateRecharge, redistributeCoverage, correlateTracks, handOverTrack and admitMember until another member assumes the role. */
    }

    #Hazard requirement MeshRadioHardwareFailureHazard {
      doc /* The physical mesh radio unit can fail from a hardware fault or wear rather than from interference, silently removing a member's coordination bearer even when RadioInterference is not present. */
    }

    #Hazard #Accepted requirement BatteryContactWearHazard {
      doc /* Accepted: the battery-swap contacts at the recharge bay wear with repeated mating cycles. Mitigated by scheduled ground-crew visual inspection at every swap, outside the software boundary the rest of this architecture addresses. */
    }
  }

  satisfy Hazards::LeaderNodeLossHazard by fleet;
  satisfy Hazards::MeshRadioHardwareFailureHazard by recoveryPoint;

  satisfy SA::Hazards::MisclassificationHazard by fleet;
  satisfy SA::Hazards::SensingDegradedUnnoticedHazard by fleet;
