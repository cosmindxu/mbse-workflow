    doc /* Logical Architecture, alternative 1 of 2 -- ground-centred coordination and command-and-control. The deciding coordination functions (sector handover, recharge rotation, coverage redistribution, deconfliction, member admission) and all command-and-control functions sit in one ground component that never flies; only the two coordination functions that are inseparable from mesh hardware itself (relaying a member's traffic, handing a track's custody to a neighbour) execute on the members. This is a genuinely different split from a member-held alternative: intelligence is centralised at the cost of dependence on the ground-station link, stated below as an accepted hazard rather than hidden. */

                doc /* holding watch over its assigned sector */

                doc /* cut off from the mesh; returning to the recovery point on its own */

    package Hazards {
        doc /* hazards introduced by this layer's own split of responsibilities, not stated above; named so as not to collide with any hazard named at SA */

        #Hazard requirement GroundCoordinationSinglePointOfFailureHazard {
            doc /* this architecture centralises sector handover, rotation, redistribution, deconfliction, admission and track correlation in one ground component; if the ground station or its link fails, the fleet loses its deciding coordination even though every member and the mesh itself remain healthy */
            subject affected : SurveillanceDroneSwarm::LA::GroundStation;
        }

        #Accepted requirement GroundCoordinationSinglePointOfFailureHazardAcceptance {
            doc /* accepted as the defining trade-off of the ground-centred alternative: centralising the deciding coordination is what buys the low reportDeliveryLatencySeconds and low onboardClassifierCostUsdPerDrone estimates below; a member-held alternative trades those away for resilience to this single point, and that is exactly the choice the customer is asked to make between the two architectures */
            subject affected : SurveillanceDroneSwarm::LA::GroundStation;
        }

        #Hazard requirement DirectiveHandoffLossOnLinkGapHazard {
            doc /* a coordination directive issued by the ground and in flight to a member when the ground-station link drops is lost before it lands on that member, leaving a sector hand-over or rotation half-completed until the link is restored */
            subject affected : SurveillanceDroneSwarm::LA::SurveillanceDrone;
        }
    }

    satisfy Hazards::GroundCoordinationSinglePointOfFailureHazard by Hazards::GroundCoordinationSinglePointOfFailureHazardAcceptance;
    satisfy Hazards::DirectiveHandoffLossOnLinkGapHazard by fleet;

    #Estimate attribute areaUnderWatchFraction :> Common::areaUnderWatchFraction {
        doc /* derived from the fleet's own arithmetic, not stated: eight of twelve members airborne at once, each covering 3.0 km2, against the worst-case 25 km2 area of interest */
    }
    assert constraint { areaUnderWatchFraction == coverageCapacity / areaOfInterestValue }

    #Estimate attribute coverageUnderMeshJammingFraction :> Common::coverageUnderMeshJammingFraction {
        doc /* derived from areaUnderWatchFraction and the jammed-link-share budget: because the deciding coordination does not depend on the mesh in this architecture, only the mesh-dependent share of coverage (relay, track hand-off) is exposed to jamming */
    }
    assert constraint { coverageUnderMeshJammingFraction == areaUnderWatchFraction * (1.0 - jammedLinkShareValue * meshDependentCoverageShare) }
